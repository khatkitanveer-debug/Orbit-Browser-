   437→    private fun closeTab(id: Long) {
   438→        val idx = tabs.indexOfFirst { it.id == id }
   439→        if (idx < 0) return
   440→        closeTabInternal(tabs[idx])
   441→        if (tabs.isEmpty()) {
   442→            finish()
   443→        } else {
   444→            selectTab(tabs[(idx - 1).coerceAtLeast(0)].id)
   445→        }
   446→    }

   448→    private fun closeTabInternal(tab: Tab) {
   449→        val idx = tabs.indexOf(tab)
   450→        if (idx < 0) return
   451→        tabs.remove(tab)
   452→        tab.webView?.apply {
   453→            stopLoading()
   454→            (parent as? ViewGroup)?.removeView(this)
   455→            destroy()
   456→        }
   457→        // NULL karna ZARURI hai — warna AI-stream/JS callbacks destroy hua WebView
   458→        // use karke crash kar sakte hain
   459→        tab.webView = null
   460→        if (tabSwitcher.visibility == View.VISIBLE) renderSwitcherCards()
   461→        refreshChrome()
   462→    }

   464→    @SuppressLint("SetJavaScriptEnabled")
   465→    private fun selectTab(id: Long) {
   466→        // firstOrNull — band ho chuke tab ka stale id kabhi app crash nahi karega
   467→        val tab = tabs.firstOrNull { it.id == id } ?: return
   468→        // outgoing tab: lastActive abhi set + thumbnail capture — switcher cards me dikhega
   469→        activeTab?.let {
   470→            it.lastActive = System.currentTimeMillis()
   471→            captureThumb(it)
   472→        }
   473→        if (tab.webView == null) {
   474→            tab.webView = buildWebView(tab)
   475→            // OPERA MODULE — evicted tab wapas aaya to URL dobara load (WebView fresh hai)
   476→            if (tab.evicted) {
   477→                tab.evicted = false
   478→                if (!tab.isInternal && tab.url != NEWTAB) tab.webView?.loadUrl(tab.url)
   479→            }
   480→        }
   481→        container.removeAllViews()
   482→        tab.webView?.let {
   483→            it.visibility = View.VISIBLE
   484→            container.addView(
   485→                it,
   486→                FrameLayout.LayoutParams(
   487→                    ViewGroup.LayoutParams.MATCH_PARENT,
   488→                    ViewGroup.LayoutParams.MATCH_PARENT,
   489→                ),
   490→            )
   491→            // EDGE MODULE — sleeping tab ko jagao
   492→            if (tab.sleeping) {
   493→                it.onResume()
   494→                tab.sleeping = false
   495→            }
   496→        }
   497→        activeTab = tab
   498→        tab.lastActive = System.currentTimeMillis()
   499→        // OPERA MODULE — RAM limit: sabse zyada purane inactive WebViews freeze/destroy
   500→        evictExcessTabs()
   501→        refreshChrome()
   502→        if (tabSwitcher.visibility == View.VISIBLE) renderSwitcherCards()
   503→        if (tab.isInternal && tab.url == NEWTAB) {
   504→            loadInternal(tab, NEWTAB)
   505→        }
   506→    }

   647→    /** web app ke shared.ts ka port — orbit:// internal pages + search detection */
   648→    private fun normalizeInput(t: String): String {
   649→        val x = t.trim()
   650→        val lower = x.lowercase()
   651→        return when {
   652→            lower.startsWith("orbit://newtab") -> NEWTAB
   653→            lower.startsWith("orbit://history") -> "orbit://history"
   654→            lower.startsWith("orbit://bookmarks") -> "orbit://bookmarks"
   655→            lower.startsWith("orbit://settings") -> "orbit://settings"
   656→            lower.startsWith("tez://") -> normalizeInput(x.replaceFirst(Regex("^tez://", RegexOption.IGNORE_CASE), "orbit://"))
   657→            lower.startsWith("http://") || lower.startsWith("https://") -> x
   658→            lower.startsWith("orbit://") -> NEWTAB
   659→            else -> {
   660→                val looksLikeUrl = !x.contains(" ") &&
   661→                    (x.contains(".") || x.contains(":")) &&
   662→                    !x.endsWith(".")
   663→                if (looksLikeUrl) "https://$x" else GOOGLE_SEARCH + java.net.URLEncoder.encode(x, "UTF-8")
   664→            }
   665→        }
   666→    }

   668→    private fun navigate(tabId: Long, url: String) {
   669→        val tab = tabs.firstOrNull { it.id == tabId } ?: return
   670→        tab.url = url
   671→        when {
   672→            url == NEWTAB -> loadInternal(tab, NEWTAB)
   673→            url.startsWith("orbit://search?q=") -> {
   674→                val q = java.net.URLDecoder.decode(url.removePrefix("orbit://search?q="), "UTF-8")
   675→                loadInternal(tab, "orbit://search?q=" + java.net.URLEncoder.encode(q, "UTF-8"))
   676→            }
   677→            url.startsWith("orbit://") -> loadInternal(tab, url)
   678→            else -> loadUrlInTab(tab, url)
   679→        }
   680→        refreshChrome()
   681→    }

   683→    @SuppressLint("SetJavaScriptEnabled")
   684→    private fun loadInternal(tab: Tab, url: String) {
   685→        tab.isInternal = true
   686→        tab.title = when {
   687→            url == NEWTAB -> "New Tab"
   688→            url.startsWith("orbit://search") -> "Orbit AI Search"
   689→            else -> "Orbit"
   690→        }
   691→        val wv = tab.webView ?: buildWebView(tab).also { tab.webView = it }
   692→        when {
   693→            url == NEWTAB -> wv.loadDataWithBaseURL("https://orbit.local/", assetHtml("newtab.html"), "text/html", "UTF-8", NEWTAB)
   694→            url.startsWith("orbit://search?q=") -> {
   695→                val q = java.net.URLDecoder.decode(url.removePrefix("orbit://search?q="), "UTF-8")
   696→                val html = assetHtml("aipage.html").replace("__QUERY__", escapeHtml(q))
   697→                wv.loadDataWithBaseURL("https://orbit.local/", html, "text/html", "UTF-8", url)
   698→                wv.postDelayed({ runAiSearch(tab, q) }, 350)
   699→            }
   700→            url == "orbit://history" -> { showHistory(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   701→            url == "orbit://bookmarks" -> { showBookmarks(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   702→            url == "orbit://settings" -> { showSettings(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   703→        }
   704→        refreshChrome()
   705→    }

   738→                override fun onError(message: String) = postJs(tab, "window.__aiError(${escapeJs(message)})")

   782→    private inner class OrbitWebViewClient(val tab: Tab) : WebViewClient() {
   783→        // BRAVE MODULE — matched ad/tracker request server tak jaane se PEHLE drop
   784→        override fun shouldInterceptRequest(view: WebView?, request: WebResourceRequest?): WebResourceResponse? {
   785→            val url = request?.url?.toString() ?: return null
   786→            if (shieldsEnabled() && AdBlocker.shouldBlock(url)) {
   787→                tab.blockedCount++
   788→                AdBlocker.countBlock()
   789→                return WebResourceResponse("text/plain", "utf-8", ByteArrayInputStream(ByteArray(0)))
   790→            }
   791→            return null
   792→        }
   793→
   794→        override fun shouldOverrideUrlLoading(view: WebView?, request: WebResourceRequest?): Boolean {
   795→            val url = request?.url ?: return false
   796→            val scheme = url.scheme?.lowercase() ?: return false
   797→            // tel:/mailto:/market: jaise external schemes — system app me bhejo
   798→            if (scheme !in setOf("http", "https", "about", "data", "javascript", "blob")) {
   799→                runCatching { startActivity(Intent(Intent.ACTION_VIEW, url)) }
   800→                return true
   801→            }
   802→            return false // http(s) WebView (Chromium) me hi khulega — real browser
   803→        }
   804→
   805→        override fun onPageStarted(view: WebView?, url: String?, favicon: Bitmap?) {
   806→            if (!url.isNullOrEmpty() && url != "https://orbit.local/" && !url.startsWith("data:")) {
   807→                tab.url = url
   808→                refreshChrome()
   809→            }
   810→        }
   811→
   812→        override fun onPageFinished(view: WebView?, url: String?) {
   813→            pullRefresh.isRefreshing = false
   814→            updateShieldBadge()
   815→            if (view == null || url.isNullOrEmpty() || tab.isInternal) return
   816→            tab.url = url
   817→            tab.title = view.title?.ifBlank { null } ?: domainOf(url)
   818→            refreshChrome()
   819→            // Incognito tab — history me KUCH nahi jata (Chrome parity)
   820→            if (!tab.isIncognito) LocalStore.addHistory(this@MainActivity, url, tab.title)
   821→        }
   822→
   823→        override fun onReceivedError(view: WebView?, request: WebResourceRequest?, error: WebResourceError?) {
   824→            // sirf main frame — subresource errors page todte nahi
   825→            if (request?.isForMainFrame == true) {
   826→                showErrorPage(view ?: return, request.url.toString(), error?.description?.toString() ?: "Page load nahi hua")
   827→            }
   828→        }
   829→    }
