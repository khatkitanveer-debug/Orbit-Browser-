   148→    private fun closeTab(id: Long) {
   149→        val idx = tabs.indexOfFirst { it.id == id }
   150→        if (idx < 0) return
   151→        val tab = tabs.removeAt(idx)
   152→        tab.webView?.apply {
   153→            stopLoading()
   154→            (parent as? ViewGroup)?.removeView(this)
   155→            destroy()
   156→        }
   157→        if (tabs.isEmpty()) {
   158→            finish()
   159→        } else {
   160→            selectTab(tabs[(idx - 1).coerceAtLeast(0)].id)
   161→        }
   162→    }

   164→    @SuppressLint("SetJavaScriptEnabled")
   165→    private fun selectTab(id: Long) {
   166→        val tab = tabs.first { it.id == id }
   167→        if (tab.webView == null) {
   168→            tab.webView = buildWebView(tab)
   169→        }
   170→        container.removeAllViews()

   176→                    ViewGroup.LayoutParams.MATCH_PARENT,

   179→            )

   182→        refreshOmnibox()
   183→        renderTabStrip()

   185→            loadInternal(tab, NEWTAB)
   186→        }
   187→    }

   253→    /** web app ke shared.ts ka port — orbit:// internal pages + search detection */
   254→    private fun normalizeInput(t: String): String {
   255→        val x = t.trim()
   256→        val lower = x.lowercase()
   257→        return when {
   258→            lower.startsWith("orbit://newtab") -> NEWTAB
   259→            lower.startsWith("orbit://history") -> "orbit://history"
   260→            lower.startsWith("orbit://bookmarks") -> "orbit://bookmarks"
   261→            lower.startsWith("orbit://settings") -> "orbit://settings"
   262→            lower.startsWith("tez://") -> normalizeInput(x.replaceFirst(Regex("^tez://", RegexOption.IGNORE_CASE), "orbit://"))
   263→            lower.startsWith("http://") || lower.startsWith("https://") -> x
   264→            lower.startsWith("orbit://") -> NEWTAB
   265→            else -> {
   266→                val looksLikeUrl = !x.contains(" ") &&
   267→                    (x.contains(".") || x.contains(":")) &&
   268→                    !x.endsWith(".")
   269→                if (looksLikeUrl) "https://$x" else GOOGLE_SEARCH + java.net.URLEncoder.encode(x, "UTF-8")
   270→            }
   271→        }
   272→    }

   274→    private fun navigate(tabId: Long, url: String) {
   275→        val tab = tabs.firstOrNull { it.id == tabId } ?: return
   276→        tab.url = url
   277→        when {
   278→            url == NEWTAB -> loadInternal(tab, NEWTAB)
   279→            url.startsWith("orbit://search?q=") -> {
   280→                val q = java.net.URLDecoder.decode(url.removePrefix("orbit://search?q="), "UTF-8")
   281→                loadInternal(tab, "orbit://search?q=" + java.net.URLEncoder.encode(q, "UTF-8"))
   282→            }
   283→            url.startsWith("orbit://") -> loadInternal(tab, url)
   284→            else -> loadUrlInTab(tab, url)
   285→        }
   286→        refreshOmnibox()
   287→        renderTabStrip()
   288→    }

   290→    @SuppressLint("SetJavaScriptEnabled")
   291→    private fun loadInternal(tab: Tab, url: String) {
   292→        tab.isInternal = true
   293→        tab.title = when {
   294→            url == NEWTAB -> "New Tab"
   295→            url.startsWith("orbit://search") -> "Orbit AI Search"
   296→            else -> "Orbit"
   297→        }
   298→        val wv = tab.webView ?: buildWebView(tab).also { tab.webView = it }
   299→        when {
   300→            url == NEWTAB -> wv.loadDataWithBaseURL("https://orbit.local/", assetHtml("newtab.html"), "text/html", "UTF-8", NEWTAB)
   301→            url.startsWith("orbit://search?q=") -> {
   302→                val q = java.net.URLDecoder.decode(url.removePrefix("orbit://search?q="), "UTF-8")
   303→                val html = assetHtml("aipage.html").replace("__QUERY__", escapeHtml(q))
   304→                wv.loadDataWithBaseURL("https://orbit.local/", html, "text/html", "UTF-8", url)
   305→                wv.postDelayed({ runAiSearch(tab, q) }, 350)
   306→            }
   307→            url == "orbit://history" -> { showHistory(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   308→            url == "orbit://bookmarks" -> { showBookmarks(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   309→            url == "orbit://settings" -> { showSettings(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   310→        }
   311→        refreshOmnibox()
   312→        renderTabStrip()
   313→    }
   314→
   315→    private fun loadUrlInTab(tab: Tab, url: String) {
   316→        tab.isInternal = false
   317→        val wv = tab.webView ?: buildWebView(tab).also { tab.webView = it }
   318→        // WebView hi Chromium hai — DIRECT load: cookies, JS, redirects, login sab
   319→        // native browser jaisa chalega. (Purana proxy-fetch approach Google jaise
   320→        // JS-heavy sites ko todta tha — browser browser ki tarah chalna chahiye.)
   321→        wv.loadUrl(url)
   322→        refreshOmnibox()
   323→    }

   334→    private fun runAiSearch(tab: Tab, query: String) {
   335→        aiJob?.interrupt()
   336→        val backend = LocalStore.getSetting(this, "aiBackend", DEFAULT_AI_BACKEND)
   337→        val wv = tab.webView ?: return
   338→        if (backend.isBlank() || backend == "unset") {
   339→            wv.evaluateJavascript("window.__aiError('Pehle Settings me apna Orbit AI backend URL set karo (menu -> Settings)')", null)
   340→            return
   341→        }
   342→        aiJob = Thread {
   343→            AiClient.search(backend, query, object : AiClient.Listener {
   344→                override fun onSources(json: String) = postJs(tab, "window.__aiSources(${escapeJs(json)})")
   345→                override fun onDelta(text: String) = postJs(tab, "window.__aiDelta(${escapeJs(text)})")
   346→                override fun onDone() = postJs(tab, "window.__aiDone()")
   347→                override fun onError(message: String) = postJs(tab, "window.__aiError(${escapeJs(message)})")
   348→            })
   349→        }.also { it.start() }
   350→    }

   353→        runOnUiThread {
   354→            if (!isFinishing) tab.webView?.evaluateJavascript(js, null)

   390→    private inner class OrbitWebViewClient(val tab: Tab) : WebViewClient() {
   391→        override fun shouldOverrideUrlLoading(view: WebView?, request: WebResourceRequest?): Boolean {
   392→            val url = request?.url ?: return false
   393→            val scheme = url.scheme?.lowercase() ?: return false
   394→            // tel:/mailto:/market: jaise external schemes — system app me bhejo
   395→            if (scheme !in setOf("http", "https", "about", "data", "javascript", "blob")) {
   396→                runCatching { startActivity(Intent(Intent.ACTION_VIEW, url)) }
   397→                return true
   398→            }
   399→            return false // http(s) WebView (Chromium) me hi khulega — real browser
   400→        }
   401→
   402→        override fun onPageStarted(view: WebView?, url: String?, favicon: Bitmap?) {
   403→            if (!url.isNullOrEmpty() && url != "https://orbit.local/" && !url.startsWith("data:")) {
   404→                tab.url = url
   405→                refreshOmnibox()
   406→            }
   407→        }
   408→
   409→        override fun onPageFinished(view: WebView?, url: String?) {
   410→            if (view == null || url.isNullOrEmpty() || tab.isInternal) return
   411→            tab.url = url
   412→            tab.title = view.title?.ifBlank { null } ?: domainOf(url)
   413→            refreshOmnibox()
   414→            renderTabStrip()
   415→            if (!privateMode) LocalStore.addHistory(this@MainActivity, url, tab.title)
   416→        }
   417→
   418→        override fun onReceivedError(view: WebView?, request: WebResourceRequest?, error: WebResourceError?) {
   419→            // sirf main frame — subresource errors page todte nahi
   420→            if (request?.isForMainFrame == true) {
   421→                showErrorPage(view ?: return, request.url.toString(), error?.description?.toString() ?: "Page load nahi hua")
   422→            }
   423→        }
   424→    }

   604→    /**
   605→     * Fully silent auto-update — launch par bhi, menu check par bhi same flow:
   606→     * update mila → background download (Orbit ke cache me) → sirf INSTALL wala
   607→     * dialog. Koi "Update karo?" puchha-woochha nahi, koi Chrome/Downloads nahi.
   608→     */
   609→    private fun runAutoUpdate(manual: Boolean) {
   610→        UpdateChecker.autoUpdate(this, manual) { info ->
   611→            runOnUiThread {
   612→                if (isFinishing) return@runOnUiThread
   613→                AlertDialog.Builder(this, R.style.OrbitDialog)
   614→                    .setTitle("Orbit ${info.versionName} ready hai")
   615→                    .setMessage("Install dabao — purana version replace ho jayega (aapka data safe rahega).")
   616→                    .setPositiveButton("Install") { _, _ ->
   617→                        UpdateChecker.cachedApk(this, info.versionName)
   618→                            ?.let { UpdateChecker.installWithStatus(this, it) }
   619→                    }
   620→                    .show()
   621→            }
   622→        }
   623→    }
