   253→   253→    /** web app ke shared.ts ka port — orbit:// internal pages + search detection */
   254→   254→    private fun normalizeInput(t: String): String {
   255→   255→        val x = t.trim()
   256→   256→        val lower = x.lowercase()
   257→   257→        return when {
   258→   258→            lower.startsWith("orbit://newtab") -> NEWTAB
   259→   259→            lower.startsWith("orbit://history") -> "orbit://history"
   260→   260→            lower.startsWith("orbit://bookmarks") -> "orbit://bookmarks"
   261→   261→            lower.startsWith("orbit://settings") -> "orbit://settings"
   262→   262→            lower.startsWith("tez://") -> normalizeInput(x.replaceFirst(Regex("^tez://", RegexOption.IGNORE_CASE), "orbit://"))
   263→   263→            lower.startsWith("http://") || lower.startsWith("https://") -> x
   264→   264→            lower.startsWith("orbit://") -> NEWTAB
   265→   265→            else -> {
   266→   266→                val looksLikeUrl = !x.contains(" ") &&
   267→   267→                    (x.contains(".") || x.contains(":")) &&
   268→   268→                    !x.endsWith(".")
   269→   269→                if (looksLikeUrl) "https://$x" else GOOGLE_SEARCH + java.net.URLEncoder.encode(x, "UTF-8")
   270→   270→            }
   271→   271→        }
   272→   272→    }

   274→   274→    private fun navigate(tabId: Long, url: String) {
   275→   275→        val tab = tabs.firstOrNull { it.id == tabId } ?: return
   276→   276→        tab.url = url
   277→   277→        when {
   278→   278→            url == NEWTAB -> loadInternal(tab, NEWTAB)
   279→   279→            url.startsWith("orbit://search?q=") -> {
   280→   280→                val q = java.net.URLDecoder.decode(url.removePrefix("orbit://search?q="), "UTF-8")
   281→   281→                loadInternal(tab, "orbit://search?q=" + java.net.URLEncoder.encode(q, "UTF-8"))
   282→   282→            }
   283→   283→            url.startsWith("orbit://") -> loadInternal(tab, url)
   284→   284→            else -> loadUrlInTab(tab, url)
   285→   285→        }
   286→   286→        refreshOmnibox()
   287→   287→        renderTabStrip()
   288→   288→    }

   290→   290→    @SuppressLint("SetJavaScriptEnabled")
   291→   291→    private fun loadInternal(tab: Tab, url: String) {
   292→   292→        tab.isInternal = true
   293→   293→        tab.title = when {
   294→   294→            url == NEWTAB -> "New Tab"
   295→   295→            url.startsWith("orbit://search") -> "Orbit AI Search"
   296→   296→            else -> "Orbit"
   297→   297→        }
   298→   298→        val wv = tab.webView ?: buildWebView(tab).also { tab.webView = it }
   299→   299→        when {
   300→   300→            url == NEWTAB -> wv.loadDataWithBaseURL("https://orbit.local/", assetHtml("newtab.html"), "text/html", "UTF-8", NEWTAB)
   301→   301→            url.startsWith("orbit://search?q=") -> {
   302→   302→                val q = java.net.URLDecoder.decode(url.removePrefix("orbit://search?q="), "UTF-8")
   303→   303→                val html = assetHtml("aipage.html").replace("__QUERY__", escapeHtml(q))
   304→   304→                wv.loadDataWithBaseURL("https://orbit.local/", html, "text/html", "UTF-8", url)
   305→   305→                wv.postDelayed({ runAiSearch(tab, q) }, 350)
   306→   306→            }
   307→   307→            url == "orbit://history" -> { showHistory(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   308→   308→            url == "orbit://bookmarks" -> { showBookmarks(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   309→   309→            url == "orbit://settings" -> { showSettings(); tab.url = NEWTAB; loadInternal(tab, NEWTAB) }
   310→   310→        }
   311→   311→        refreshOmnibox()
   312→   312→        renderTabStrip()
   313→   313→    }
   314→   314→
   315→   315→    private fun loadUrlInTab(tab: Tab, url: String) {
   316→   316→        tab.isInternal = false
   317→   317→        val wv = tab.webView ?: buildWebView(tab).also { tab.webView = it }
   318→   318→        // WebView hi Chromium hai — DIRECT load: cookies, JS, redirects, login sab
   319→   319→        // native browser jaisa chalega. (Purana proxy-fetch approach Google jaise
   320→   320→        // JS-heavy sites ko todta tha — browser browser ki tarah chalna chahiye.)
   321→   321→        wv.loadUrl(url)
   322→   322→        refreshOmnibox()
   323→   323→    }

   334→   334→    private fun runAiSearch(tab: Tab, query: String) {
   335→   335→        aiJob?.interrupt()
   336→   336→        val backend = LocalStore.getSetting(this, "aiBackend", DEFAULT_AI_BACKEND)
   337→   337→        val wv = tab.webView ?: return
   338→   338→        if (backend.isBlank() || backend == "unset") {
   339→   339→            wv.evaluateJavascript("window.__aiError('Pehle Settings me apna Orbit AI backend URL set karo (menu -> Settings)')", null)
   340→   340→            return
   341→   341→        }
   342→   342→        aiJob = Thread {
   343→   343→            AiClient.search(backend, query, object : AiClient.Listener {
   344→   344→                override fun onSources(json: String) = postJs(tab, "window.__aiSources(${escapeJs(json)})")
   345→   345→                override fun onDelta(text: String) = postJs(tab, "window.__aiDelta(${escapeJs(text)})")
   346→   346→                override fun onDone() = postJs(tab, "window.__aiDone()")
   347→   347→                override fun onError(message: String) = postJs(tab, "window.__aiError(${escapeJs(message)})")
   348→   348→            })
   349→   349→        }.also { it.start() }
   350→   350→    }

   352→   352→    private fun postJs(tab: Tab, js: String) {
   353→   353→        runOnUiThread {
   354→   354→            if (!isFinishing) tab.webView?.evaluateJavascript(js, null)
   355→   355→        }
   356→   356→    }

   390→   390→    private inner class OrbitWebViewClient(val tab: Tab) : WebViewClient() {
   391→   391→        override fun shouldOverrideUrlLoading(view: WebView?, request: WebResourceRequest?): Boolean {
   392→   392→            val url = request?.url ?: return false
   393→   393→            val scheme = url.scheme?.lowercase() ?: return false
   394→   394→            // tel:/mailto:/market: jaise external schemes — system app me bhejo
   395→   395→            if (scheme !in setOf("http", "https", "about", "data", "javascript", "blob")) {
   396→   396→                runCatching { startActivity(Intent(Intent.ACTION_VIEW, url)) }
   397→   397→                return true
   398→   398→            }
   399→   399→            return false // http(s) WebView (Chromium) me hi khulega — real browser
   400→   400→        }
   401→   401→
   402→   402→        override fun onPageStarted(view: WebView?, url: String?, favicon: Bitmap?) {
   403→   403→            if (!url.isNullOrEmpty() && url != "https://orbit.local/" && !url.startsWith("data:")) {
   404→   404→                tab.url = url
   405→   405→                refreshOmnibox()
   406→   406→            }
   407→   407→        }
   408→   408→
   409→   409→        override fun onPageFinished(view: WebView?, url: String?) {
   410→   410→            if (view == null || url.isNullOrEmpty() || tab.isInternal) return
   411→   411→            tab.url = url
   412→   412→            tab.title = view.title?.ifBlank { null } ?: domainOf(url)
   413→   413→            refreshOmnibox()
   414→   414→            renderTabStrip()
   415→   415→            if (!privateMode) LocalStore.addHistory(this@MainActivity, url, tab.title)
   416→   416→        }
   417→   417→
   418→   418→        override fun onReceivedError(view: WebView?, request: WebResourceRequest?, error: WebResourceError?) {
   419→   419→            // sirf main frame — subresource errors page todte nahi
   420→   420→            if (request?.isForMainFrame == true) {
   421→   421→                showErrorPage(view ?: return, request.url.toString(), error?.description?.toString() ?: "Page load nahi hua")
   422→   422→            }
   423→   423→        }
   424→   424→    }

   605→   605→     * Fully silent auto-update — launch par bhi, menu check par bhi same flow:
   606→   606→     * update mila → background download (Orbit ke cache me) → sirf INSTALL wala
   607→   607→     * dialog. Koi "Update karo?" puchha-woochha nahi, koi Chrome/Downloads nahi.
   608→   608→     */
   609→   609→    private fun runAutoUpdate(manual: Boolean) {
   610→   610→        UpdateChecker.autoUpdate(this, manual) { info ->
   611→   611→            runOnUiThread {
   612→   612→                if (isFinishing) return@runOnUiThread
   613→   613→                AlertDialog.Builder(this, R.style.OrbitDialog)
   614→   614→                    .setTitle("Orbit ${info.versionName} ready hai")
   615→   615→                    .setMessage("Install dabao — purana version replace ho jayega (aapka data safe rahega).")
   616→   616→                    .setPositiveButton("Install") { _, _ ->
   617→   617→                        UpdateChecker.cachedApk(this, info.versionName)
   618→   618→                            ?.let { UpdateChecker.installWithStatus(this, it) }
   619→   619→                    }
   620→   620→                    .show()
   621→   621→            }
   622→   622→        }
   623→   623→    }
